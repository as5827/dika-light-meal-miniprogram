server/intent.js —— num() 的一次真实修复

第一版：忘了挡 null

const num = (v, min, max) => {
  const n = Number(v);          // Number(null) === 0，不是 NaN
  if (!Number.isFinite(n)) return null;
  return Math.min(max, Math.max(min, Math.round(n)));
};

// 结果：模型没给蛋白时，条件里多出「蛋白 ≥ 0」「价格 ≤ ¥1」，
//       筛选结果被这几条假条件带偏，自测里才暴露出来。

修复后：先挡空值，再转数字

const num = (v, min, max) => {
  if (v === null || v === undefined || v === '') return null;  // 注意 Number(null) === 0，必须先挡掉
  const n = Number(v);
  if (!Number.isFinite(n)) return null;
  return Math.min(max, Math.max(min, Math.round(n)));
};
